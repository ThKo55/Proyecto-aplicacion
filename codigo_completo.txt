namespace AorusMarket.AccesoDatos

    public class CategoriaDatos
    {
        // Método para listar las categorías activas en la grilla
        public List<Categoria> Listar()
        {
            List<Categoria> lista = new List<Categoria>();

            // El bloque 'using' asegura que la conexión se cierre y libere automáticamente
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                // Solo traemos las que tienen activo = 1 (baja lógica)
                string query = "SELECT id_categoria, nombre, descripcion FROM categoria WHERE activo = 1";
                SqlCommand cmd = new SqlCommand(query, oConexion);
                cmd.CommandType = CommandType.Text;

                try
                {
                    oConexion.Open();
                    using (SqlDataReader dr = cmd.ExecuteReader())
                    {
                        while (dr.Read())
                        {
                            lista.Add(new Categoria()
                            {
                                IdCategoria = Convert.ToInt32(dr["id_categoria"]),
                                Nombre = dr["nombre"].ToString(),
                                Descripcion = dr["descripcion"].ToString()
                            });
                        }
                    }
                }
                catch (Exception ex)
                {
                    // Si hay error, devolvemos una lista vacía y lo registramos
                    lista = new List<Categoria>();
                    Console.WriteLine("Error en Listar Categorias: " + ex.Message);
                }
            }
            return lista;
        }

        // Método para Insertar una nueva categoría
        public bool Insertar(Categoria obj)
        {
            bool respuesta = false;
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                string query = "INSERT INTO categoria (nombre, descripcion) VALUES (@nombre, @descripcion)";
                SqlCommand cmd = new SqlCommand(query, oConexion);
                cmd.Parameters.AddWithValue("@nombre", obj.Nombre);
                cmd.Parameters.AddWithValue("@descripcion", obj.Descripcion);

                try
                {
                    oConexion.Open();
                    respuesta = cmd.ExecuteNonQuery() > 0; // Si afectó a más de 0 filas, fue exitoso
                }
                catch (Exception ex)
                {
                    respuesta = false;
                    Console.WriteLine("Error al insertar: " + ex.Message);
                }
            }
            return respuesta;
        }

        // Método para Editar una categoría existente
        public bool Editar(Categoria obj)
        {
            bool respuesta = false;
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                string query = "UPDATE categoria SET nombre = @nombre, descripcion = @descripcion WHERE id_categoria = @id";
                SqlCommand cmd = new SqlCommand(query, oConexion);
                cmd.Parameters.AddWithValue("@nombre", obj.Nombre);
                cmd.Parameters.AddWithValue("@descripcion", obj.Descripcion);
                cmd.Parameters.AddWithValue("@id", obj.IdCategoria);

                try
                {
                    oConexion.Open();
                    respuesta = cmd.ExecuteNonQuery() > 0;
                }
                catch (Exception ex)
                {
                    respuesta = false;
                    Console.WriteLine("Error al editar: " + ex.Message);
                }
            }
            return respuesta;
        }

        // Método para Eliminar (Baja Lógica)
        public bool Eliminar(int id)
        {
            bool respuesta = false;
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                // NO hacemos DELETE, hacemos UPDATE del estado activo a 0
                string query = "UPDATE categoria SET activo = 0 WHERE id_categoria = @id";

                cmd.Parameters.AddWithValue("@id", id);

                try
                {
                    oConexion.Open();
                    respuesta = cmd.ExecuteNonQuery() > 0;
                }
                catch (Exception ex)
                {
                    respuesta = false;
                    Console.WriteLine("Error al eliminar: " + ex.Message);
                }
            }
            return respuesta;

    public class ClienteDatos
    {
        public List<Cliente> Listar(string filtroDni = "")
        {
            List<Cliente> lista = new List<Cliente>();
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                string query = @"
                    SELECT c.id_cliente, c.dni, c.cuil_cuit, c.nombre, c.apellido, c.email, c.telefono, c.fecha_nacimiento, 
                           c.id_direccion, d.calle, d.altura
                    FROM cliente c
                    INNER JOIN direccion d ON c.id_direccion = d.id_direccion
                    WHERE c.activo = 1";

                if (!string.IsNullOrEmpty(filtroDni)) query += " AND c.dni LIKE @filtro";

                SqlCommand cmd = new SqlCommand(query, oConexion);
                if (!string.IsNullOrEmpty(filtroDni)) cmd.Parameters.AddWithValue("@filtro", "%" + filtroDni + "%");

                try
                {
                    oConexion.Open();
                    using (SqlDataReader dr = cmd.ExecuteReader())
                    {
                        while (dr.Read())
                        {
                            lista.Add(new Cliente()
                            {
                                IdCliente = Convert.ToInt32(dr["id_cliente"]),
                                Dni = dr["dni"].ToString(),
                                CuilCuit = dr["cuil_cuit"].ToString(),
                                Nombre = dr["nombre"].ToString(),
                                Apellido = dr["apellido"].ToString(),
                                Email = dr["email"].ToString(),
                                Telefono = dr["telefono"].ToString(),
                                FechaNacimiento = dr["fecha_nacimiento"] != DBNull.Value ? Convert.ToDateTime(dr["fecha_nacimiento"]) : (DateTime?)null,
                                IdDireccion = Convert.ToInt32(dr["id_direccion"]),
                                Calle = dr["calle"].ToString(),
                                Altura = dr["altura"].ToString()
                            });
                        }
                    }
                }
                catch (Exception ex) { Console.WriteLine("Error: " + ex.Message); }
            }
            return lista;
        }

        public bool Insertar(Cliente obj)
        {
            bool respuesta = false;
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                oConexion.Open();
                SqlTransaction tx = oConexion.BeginTransaction();
                try
                {
                    string qDir = "INSERT INTO direccion (calle, altura) OUTPUT INSERTED.id_direccion VALUES (@calle, @altura)";
                    SqlCommand cmdDir = new SqlCommand(qDir, oConexion, tx);
                    cmdDir.Parameters.AddWithValue("@calle", obj.Calle ?? "");
                    cmdDir.Parameters.AddWithValue("@altura", obj.Altura ?? "");
                    int idDir = Convert.ToInt32(cmdDir.ExecuteScalar());

                    string qCli = "INSERT INTO cliente (dni, cuil_cuit, nombre, apellido, email, telefono, fecha_nacimiento, id_direccion) VALUES (@dni, @cuil, @nom, @ape, @email, @tel, @fec, @iddir)";
                    SqlCommand cmdCli = new SqlCommand(qCli, oConexion, tx);
                    cmdCli.Parameters.AddWithValue("@dni", obj.Dni ?? "");
                    cmdCli.Parameters.AddWithValue("@cuil", obj.CuilCuit ?? "");
                    cmdCli.Parameters.AddWithValue("@nom", obj.Nombre);
                    cmdCli.Parameters.AddWithValue("@ape", obj.Apellido);
                    cmdCli.Parameters.AddWithValue("@email", obj.Email ?? "");
                    cmdCli.Parameters.AddWithValue("@tel", obj.Telefono ?? "");
                    cmdCli.Parameters.AddWithValue("@fec", obj.FechaNacimiento.HasValue ? (object)obj.FechaNacimiento.Value : DBNull.Value);
                    cmdCli.Parameters.AddWithValue("@iddir", idDir);

                    cmdCli.ExecuteNonQuery();
                    tx.Commit();
                    respuesta = true;
                }
                catch { tx.Rollback(); }
            }
            return respuesta;
        }

        public bool Editar(Cliente obj)
        {
            bool respuesta = false;
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                oConexion.Open();
                SqlTransaction tx = oConexion.BeginTransaction();
                try
                {
                    string qDir = "UPDATE direccion SET calle = @calle, altura = @altura WHERE id_direccion = @idDir";
                    SqlCommand cmdDir = new SqlCommand(qDir, oConexion, tx);
                    cmdDir.Parameters.AddWithValue("@calle", obj.Calle ?? "");
                    cmdDir.Parameters.AddWithValue("@altura", obj.Altura ?? "");
                    cmdDir.Parameters.AddWithValue("@idDir", obj.IdDireccion);
                    cmdDir.ExecuteNonQuery();

                    string qCli = "UPDATE cliente SET dni=@dni, cuil_cuit=@cuil, nombre=@nom, apellido=@ape, email=@email, telefono=@tel, fecha_nacimiento=@fec WHERE id_cliente=@id";
                    SqlCommand cmdCli = new SqlCommand(qCli, oConexion, tx);
                    cmdCli.Parameters.AddWithValue("@dni", obj.Dni ?? "");
                    cmdCli.Parameters.AddWithValue("@cuil", obj.CuilCuit ?? "");
                    cmdCli.Parameters.AddWithValue("@nom", obj.Nombre);
                    cmdCli.Parameters.AddWithValue("@ape", obj.Apellido);
                    cmdCli.Parameters.AddWithValue("@email", obj.Email ?? "");
                    cmdCli.Parameters.AddWithValue("@tel", obj.Telefono ?? "");
                    cmdCli.Parameters.AddWithValue("@fec", obj.FechaNacimiento.HasValue ? (object)obj.FechaNacimiento.Value : DBNull.Value);
                    cmdCli.Parameters.AddWithValue("@id", obj.IdCliente);

                    cmdCli.ExecuteNonQuery();
                    tx.Commit();
                    respuesta = true;
                }
                catch { tx.Rollback(); }
            }
            return respuesta;
        }

        public bool Eliminar(int id)
        {
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                SqlCommand cmd = new SqlCommand("UPDATE cliente SET activo = 0 WHERE id_cliente = @id", oConexion);

                oConexion.Open();
                return cmd.ExecuteNonQuery() > 0;
            }
        }

        // 2. Método para obtener las métricas de las Tarjetas Superiores
        public MetricasDashboard ObtenerMetricas(int idSucursal, DateTime desde, DateTime hasta)
        {
            MetricasDashboard metricas = new MetricasDashboard() { TotalVendido = 0, CantidadVentas = 0, ProductoMasVendido = "-" };

            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                // Consulta compleja: Suma totales, cuenta ventas y usa una subconsulta para sacar el producto más vendido
                // (@idSucursal = 0) funciona como un "comodín" para traer todas las sucursales
                string query = @"
                    SELECT 
                        ISNULL(SUM(total), 0) AS TotalVendido,
                        COUNT(id_venta) AS CantidadVentas,
                        ISNULL((
                            SELECT TOP 1 p.nombre 
                            FROM detalleVenta dv 
                            INNER JOIN producto p ON dv.id_producto = p.id_producto
                            INNER JOIN venta v2 ON v2.id_venta = dv.id_venta
                            WHERE v2.estado = 1 
                              AND v2.fecha >= @desde AND v2.fecha <= @hasta
                              AND (@idSucursal = 0 OR v2.id_sucursal = @idSucursal)
                            GROUP BY p.nombre 
                            ORDER BY SUM(dv.cantidad) DESC
                        ), '-') AS ProductoTop
                    FROM venta 
                    WHERE estado = 1 
                      AND fecha >= @desde AND fecha <= @hasta
                      AND (@idSucursal = 0 OR id_sucursal = @idSucursal)";

                SqlCommand cmd = new SqlCommand(query, oConexion);
                cmd.Parameters.AddWithValue("@idSucursal", idSucursal);
                cmd.Parameters.AddWithValue("@desde", desde);
                cmd.Parameters.AddWithValue("@hasta", hasta);

                try
                {
                    oConexion.Open();
                    using (SqlDataReader dr = cmd.ExecuteReader())
                    {
                        if (dr.Read())
                        {
                            metricas.TotalVendido = Convert.ToDecimal(dr["TotalVendido"]);
                            metricas.CantidadVentas = Convert.ToInt32(dr["CantidadVentas"]);
                            metricas.ProductoMasVendido = dr["ProductoTop"].ToString();
                        }
                    }
                }
                catch (Exception ex)
                {
                    Console.WriteLine("Error en Métricas: " + ex.Message);
                }
            }
            return metricas;
        }

        // 3. Método para listar las ventas recientes en la grilla
        public List<VentaReciente> ObtenerVentasRecientes(int idSucursal, DateTime desde, DateTime hasta)
        {
            List<VentaReciente> lista = new List<VentaReciente>();
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                // Unimos Venta, Sucursal y Cliente. 
                // Usamos ISNULL en el cliente para mostrar 'Consumidor Final' si la venta no tuvo cliente.
                string query = @"
                    SELECT 
                        v.id_venta, v.fecha, s.nombre AS Sucursal,
                        ISNULL(c.nombre + ' ' + c.apellido, 'Consumidor Final') AS Cliente,
                        v.total, v.metodo_pago,
                        CASE v.estado WHEN 1 THEN 'Confirmada' WHEN 2 THEN 'Anulada' ELSE 'Desconocido' END AS Estado
                    FROM venta v
                    INNER JOIN sucursal s ON v.id_sucursal = s.id_sucursal
                    LEFT JOIN cliente c ON v.id_cliente = c.id_cliente
                    WHERE v.fecha >= @desde AND v.fecha <= @hasta
                      AND (@idSucursal = 0 OR v.id_sucursal = @idSucursal)
                    ORDER BY v.fecha DESC";

                SqlCommand cmd = new SqlCommand(query, oConexion);
                cmd.Parameters.AddWithValue("@idSucursal", idSucursal);
                cmd.Parameters.AddWithValue("@desde", desde);
                cmd.Parameters.AddWithValue("@hasta", hasta);

                try
                {
                    oConexion.Open();
                    using (SqlDataReader dr = cmd.ExecuteReader())
                    {
                        while (dr.Read())
                        {
                            lista.Add(new VentaReciente()
                            {
                                IdVenta = Convert.ToInt32(dr["id_venta"]),
                                Fecha = Convert.ToDateTime(dr["fecha"]),
                                Sucursal = dr["Sucursal"].ToString(),
                                Cliente = dr["Cliente"].ToString(),
                                Total = Convert.ToDecimal(dr["total"]),
                                MetodoPago = dr["metodo_pago"].ToString(),
                                Estado = dr["Estado"].ToString()
                            });
                        }
                    }
                }
                catch (Exception ex)
                {
                    Console.WriteLine("Error al listar ventas recientes: " + ex.Message);
                }
            }
            return lista;
        }
    }

namespace AorusMarket.AccesoDatos
{
    public class ReporteDatos
    {
        // 1. Recaudación por Medio de Pago
        public List<ReporteMedioPago> RecaudacionPorMedioPago(DateTime desde, DateTime hasta)
        {
            List<ReporteMedioPago> lista = new List<ReporteMedioPago>();
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                string query = @"SELECT metodo_pago, SUM(total) as Total 
                                 FROM venta 
                                 WHERE estado = 1 AND CAST(fecha AS DATE) BETWEEN @desde AND @hasta 
                                 GROUP BY metodo_pago";
                SqlCommand cmd = new SqlCommand(query, oConexion);
                cmd.Parameters.AddWithValue("@desde", desde.Date);
                cmd.Parameters.AddWithValue("@hasta", hasta.Date);
                try
                {
                    oConexion.Open();
                    using (SqlDataReader dr = cmd.ExecuteReader())
                    {
                        while (dr.Read())
                        {
                            lista.Add(new ReporteMedioPago()
                            {
                                MedioPago = dr["metodo_pago"].ToString(),
                                TotalRecaudado = Convert.ToDecimal(dr["Total"])
                            });
                        }
                    }
                }
                catch (Exception ex) { Console.WriteLine(ex.Message); }
            }
            return lista;

        public bool Guardar(StockSucursal obj, int idUsuarioLogueado)
        {
            bool respuesta = false;
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                oConexion.Open();
                SqlTransaction transaccion = oConexion.BeginTransaction();
                try
                {
                    string checkQuery = "SELECT id_stock_sucursal, cantidad FROM stock_sucursal WHERE id_producto = @idProd AND id_sucursal = @idSuc";
                    SqlCommand cmdCheck = new SqlCommand(checkQuery, oConexion, transaccion);
                    cmdCheck.Parameters.AddWithValue("@idProd", obj.IdProducto);
                    cmdCheck.Parameters.AddWithValue("@idSuc", obj.IdSucursal);

                    int idStockExistente = 0;
                    int cantidadAnterior = 0;

                    using (SqlDataReader dr = cmdCheck.ExecuteReader())
                    {
                        if (dr.Read())
                        {
                            idStockExistente = Convert.ToInt32(dr["id_stock_sucursal"]);
                            cantidadAnterior = Convert.ToInt32(dr["cantidad"]);
                        }
                    }

                    int idStockAfectado = 0;
                    int variacion = 0;

                    if (idStockExistente > 0)
                    {
                        string updateQuery = "UPDATE stock_sucursal SET cantidad = @cant, precio = @prec, stock_minimo = @min WHERE id_stock_sucursal = @idStock";
                        SqlCommand cmdUpdate = new SqlCommand(updateQuery, oConexion, transaccion);
                        cmdUpdate.Parameters.AddWithValue("@cant", obj.Cantidad);
                        cmdUpdate.Parameters.AddWithValue("@prec", obj.Precio);
                        cmdUpdate.Parameters.AddWithValue("@min", obj.StockMinimo);
                        cmdUpdate.Parameters.AddWithValue("@idStock", idStockExistente);
                        cmdUpdate.ExecuteNonQuery();

                        idStockAfectado = idStockExistente;
                        variacion = obj.Cantidad - cantidadAnterior;
                    }
                    else
                    {
                        string insertQuery = @"
                            INSERT INTO stock_sucursal (id_producto, id_sucursal, cantidad, precio, stock_minimo) 
                            OUTPUT INSERTED.id_stock_sucursal
                            VALUES (@idProd, @idSuc, @cant, @prec, @min)";
                        SqlCommand cmdInsert = new SqlCommand(insertQuery, oConexion, transaccion);
                        cmdInsert.Parameters.AddWithValue("@idProd", obj.IdProducto);
                        cmdInsert.Parameters.AddWithValue("@idSuc", obj.IdSucursal);
                        cmdInsert.Parameters.AddWithValue("@cant", obj.Cantidad);
                        cmdInsert.Parameters.AddWithValue("@prec", obj.Precio);
                        cmdInsert.Parameters.AddWithValue("@min", obj.StockMinimo);

                        idStockAfectado = Convert.ToInt32(cmdInsert.ExecuteScalar());
                        variacion = obj.Cantidad;
                    }

                    if (variacion != 0)
                    {
                        string movQuery = @"
                            INSERT INTO movimiento_stock (id_stock_sucursal, cantidad_variacion, tipo_movimiento, id_usuario)
                            VALUES (@idStock, @var, 'Ajuste Manual', @idUsuario)";
                        SqlCommand cmdMov = new SqlCommand(movQuery, oConexion, transaccion);
                        cmdMov.Parameters.AddWithValue("@idStock", idStockAfectado);
                        cmdMov.Parameters.AddWithValue("@var", variacion);
                        cmdMov.Parameters.AddWithValue("@idUsuario", idUsuarioLogueado);
                        cmdMov.ExecuteNonQuery();
                    }

                    transaccion.Commit();
                    respuesta = true;
                }
                catch { transaccion.Rollback(); }
            }
            return respuesta;
        }

            }
        }

        public Usuario Autenticar(string email, string password)
        {
            Usuario obj = null;
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                // Quitamos "u.password = @pass" del WHERE para validarlo desde C# con BCrypt
                string query = @"
                    SELECT u.id_usuario, u.nombre, u.apellido, u.email, u.password, u.id_perfil, u.id_sucursal,
                           ISNULL(p.nombre, '') AS NombrePerfil, ISNULL(s.nombre, '') AS NombreSucursal
                    FROM usuario u
                    LEFT JOIN perfil p ON u.id_perfil = p.id_perfil
                    LEFT JOIN sucursal s ON u.id_sucursal = s.id_sucursal
                    WHERE u.email = @email AND u.activo = 1";

                SqlCommand cmd = new SqlCommand(query, oConexion);
                cmd.Parameters.AddWithValue("@email", email);
                try
                {
                    oConexion.Open();
                    using (SqlDataReader dr = cmd.ExecuteReader())
                    {
                        if (dr.Read())
                        {
                            string passGuardada = dr["password"].ToString();
                            bool passwordValida = false;

                            // Verificamos si la contraseña está encriptada con BCrypt ($2a$, $2b$, $2y$)
                            if (passGuardada.StartsWith("$2"))
                            {
                                passwordValida = BCrypt.Net.BCrypt.Verify(password, passGuardada);
                            }
                            else
                            {
                                // Fallback para usuarios viejos con texto plano (ej: '1234')
                                passwordValida = (password == passGuardada);
                            }

                            if (passwordValida)
                            {
                                obj = new Usuario()
                                {
                                    IdUsuario = Convert.ToInt32(dr["id_usuario"]),
                                    Nombre = dr["nombre"].ToString(),
                                    Apellido = dr["apellido"].ToString(),
                                    Email = dr["email"].ToString(),
                                    IdPerfil = Convert.ToInt32(dr["id_perfil"]),
                                    IdSucursal = Convert.ToInt32(dr["id_sucursal"]),
                                    NombrePerfil = dr["NombrePerfil"].ToString(),
                                    NombreSucursal = dr["NombreSucursal"].ToString()
                                };
                            }
                        }
                    }
                }
                catch (Exception ex)
                {
                    MessageBox.Show("Error en Autenticar: " + ex.Message);
                }
            }
            return obj;
        }

                                IdSucursal = dr["id_sucursal"] != DBNull.Value ? Convert.ToInt32(dr["id_sucursal"]) : 0,
                                NombrePerfil = dr["NombrePerfil"] != DBNull.Value ? dr["NombrePerfil"].ToString() : "",
                                NombreSucursal = dr["NombreSucursal"] != DBNull.Value ? dr["NombreSucursal"].ToString() : "",

            {
                oConexion.Open();
                SqlTransaction tx = oConexion.BeginTransaction();

                {
                    string qDir = "INSERT INTO direccion (calle, altura) OUTPUT INSERTED.id_direccion VALUES (@calle, @altura)";

                    cmdDir.Parameters.AddWithValue("@calle", obj.Calle ?? "");
                    cmdDir.Parameters.AddWithValue("@altura", obj.Altura ?? "");
                    int idDir = Convert.ToInt32(cmdDir.ExecuteScalar());

                    string qUsr = "INSERT INTO usuario (dni, nombre, apellido, email, password, telefono, fecha_nacimiento, id_perfil, id_sucursal, id_direccion) VALUES (@dni, @nom, @ape, @email, @pass, @tel, @fec, @perfil, @sucursal, @iddir)";

            {
                const string query = @"
                    SELECT p.id_producto, ss.id_stock_sucursal, p.codigo_barras, p.nombre, ss.precio, ss.cantidad
                    FROM producto p
                    INNER JOIN stock_sucursal ss ON p.id_producto = ss.id_producto
                    WHERE ss.id_sucursal = @idSucursal 
                      AND p.activo = 1 
                      AND (p.codigo_barras = @textoExacto OR p.nombre LIKE @textoParcial)";

                SqlCommand cmd = new SqlCommand(query, oConexion);
                cmd.Parameters.AddWithValue("@idSucursal", idSucursal);
                cmd.Parameters.AddWithValue("@textoExacto", texto.Trim());
                cmd.Parameters.AddWithValue("@textoParcial", "%" + texto.Trim() + "%");

                try
                {
                    oConexion.Open();
                    using (SqlDataReader dr = cmd.ExecuteReader())
                    {
                        while (dr.Read())
                        {
                            lista.Add(new BusquedaProducto()
                            {
                                IdProducto = Convert.ToInt32(dr["id_producto"]),
                                IdStockSucursal = Convert.ToInt32(dr["id_stock_sucursal"]),
                                CodigoBarras = dr["codigo_barras"] != DBNull.Value ? dr["codigo_barras"].ToString() : "",
                                Nombre = dr["nombre"].ToString(),

                                StockDisponible = Convert.ToInt32(dr["cantidad"])

                }

        public bool ConfirmarVenta(Venta objVenta)
        {
            bool respuesta = false;
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                oConexion.Open();
                SqlTransaction transaccion = oConexion.BeginTransaction();
                try
                {
                    const string queryVenta = @"
                        INSERT INTO venta (total, metodo_pago, id_cliente, id_usuario, id_sucursal) 
                        OUTPUT INSERTED.id_venta
                        VALUES (@total, @metodo, @cliente, @usuario, @sucursal)";

                    SqlCommand cmdVenta = new SqlCommand(queryVenta, oConexion, transaccion);
                    cmdVenta.Parameters.AddWithValue("@total", objVenta.Total);
                    cmdVenta.Parameters.AddWithValue("@metodo", objVenta.MetodoPago);
                    cmdVenta.Parameters.AddWithValue("@cliente", objVenta.IdCliente.HasValue ? (object)objVenta.IdCliente.Value : DBNull.Value);
                    cmdVenta.Parameters.AddWithValue("@usuario", objVenta.IdUsuario);
                    cmdVenta.Parameters.AddWithValue("@sucursal", objVenta.IdSucursal);

                    int idVentaGenerado = Convert.ToInt32(cmdVenta.ExecuteScalar());
                    objVenta.IdVenta = idVentaGenerado;

                    const string paramCant = "@cant";

                    foreach (var item in objVenta.Detalles)
                    {
                        const string queryDetalle = "INSERT INTO detalleVenta (id_venta, id_producto, cantidad, precio_unitario, subtotal) VALUES (@idVenta, @idProd, @cant, @precio, @subt)";
                        SqlCommand cmdDetalle = new SqlCommand(queryDetalle, oConexion, transaccion);
                        cmdDetalle.Parameters.AddWithValue("@idVenta", idVentaGenerado);
                        cmdDetalle.Parameters.AddWithValue("@idProd", item.IdProducto);
                        cmdDetalle.Parameters.AddWithValue(paramCant, item.Cantidad);
                        cmdDetalle.Parameters.AddWithValue("@precio", item.PrecioUnitario);
                        cmdDetalle.Parameters.AddWithValue("@subt", item.Subtotal);
                        cmdDetalle.ExecuteNonQuery();

                        const string queryStock = @"
                            UPDATE stock_sucursal SET cantidad = cantidad - @cant 
                            OUTPUT INSERTED.id_stock_sucursal
                            WHERE id_producto = @idProd AND id_sucursal = @sucursal AND cantidad >= @cant";
                        SqlCommand cmdStock = new SqlCommand(queryStock, oConexion, transaccion);
                        cmdStock.Parameters.AddWithValue(paramCant, item.Cantidad);
                        cmdStock.Parameters.AddWithValue("@idProd", item.IdProducto);
                        cmdStock.Parameters.AddWithValue("@sucursal", objVenta.IdSucursal);

                        object resStock = cmdStock.ExecuteScalar();
                        // SOLUCIÓN IDE0270 y S112: Manejo limpio de nulo sin 'throw new Exception'
                        int idStockSucursal = Convert.ToInt32(resStock ?? throw new InvalidOperationException("Stock insuficiente durante la confirmación."));

                        const string queryAuditoria = "INSERT INTO movimiento_stock (id_stock_sucursal, cantidad_variacion, tipo_movimiento, id_usuario) VALUES (@idStock, @variacion, 'Venta', @usuario)";
                        SqlCommand cmdAuditoria = new SqlCommand(queryAuditoria, oConexion, transaccion);
                        cmdAuditoria.Parameters.AddWithValue("@idStock", idStockSucursal);
                        cmdAuditoria.Parameters.AddWithValue("@variacion", -item.Cantidad);
                        cmdAuditoria.Parameters.AddWithValue("@usuario", objVenta.IdUsuario);
                        cmdAuditoria.ExecuteNonQuery();
                    }

                    transaccion.Commit();
                    respuesta = true;
                }
                catch (Exception ex)
                {
                    transaccion.Rollback();
                    respuesta = false;
                    Console.WriteLine("Error crítico en Venta: " + ex.Message);
                }
            }
            return respuesta;
        }

        public List<VentaHistorial> ListarHistorial(DateTime fechaDesde, DateTime fechaHasta, int idUsuarioFiltro)
        {
            List<VentaHistorial> lista = new List<VentaHistorial>();
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                string query = @"
                    SELECT v.id_venta, v.fecha, 
                           ISNULL(c.nombre + ' ' + c.apellido, 'Consumidor Final') AS Cliente,
                           s.nombre AS Sucursal, 
                           u.nombre + ' ' + u.apellido AS Cajero,
                           v.total, v.metodo_pago, v.estado
                    FROM venta v
                    LEFT JOIN cliente c ON v.id_cliente = c.id_cliente
                    INNER JOIN sucursal s ON v.id_sucursal = s.id_sucursal
                    INNER JOIN usuario u ON v.id_usuario = u.id_usuario
                    WHERE CAST(v.fecha AS DATE) BETWEEN @desde AND @hasta ";

                if (idUsuarioFiltro > 0) query += " AND v.id_usuario = @idUsr";
                query += " ORDER BY v.fecha DESC";

                SqlCommand cmd = new SqlCommand(query, oConexion);
                cmd.Parameters.AddWithValue("@desde", fechaDesde.Date);
                cmd.Parameters.AddWithValue("@hasta", fechaHasta.Date);
                if (idUsuarioFiltro > 0) cmd.Parameters.AddWithValue("@idUsr", idUsuarioFiltro);

                try
                {
                    oConexion.Open();
                    using (SqlDataReader dr = cmd.ExecuteReader())
                    {
                        while (dr.Read())
                        {
                            lista.Add(new VentaHistorial()
                            {
                                IdVenta = Convert.ToInt32(dr["id_venta"]),
                                Fecha = Convert.ToDateTime(dr["fecha"]).ToString("dd/MM/yyyy HH:mm"),
                                Cliente = dr["Cliente"].ToString(),
                                Sucursal = dr["Sucursal"].ToString(),
                                Cajero = dr["Cajero"].ToString(),
                                Total = Convert.ToDecimal(dr["total"]),
                                MetodoPago = dr["metodo_pago"].ToString(),
                                Estado = Convert.ToInt32(dr["estado"]) == 1 ? "Confirmada" : "Anulada"
                            });
                        }
                    }
                }
                catch (Exception ex) { Console.WriteLine("Error al listar historial: " + ex.Message); }
            }
            return lista;
        }

        public bool AnularVenta(int idVenta, int idUsuarioAuditoria)
        {
            bool respuesta = false;
            using (SqlConnection oConexion = Conexion.ObtenerConexion())
            {
                oConexion.Open();
                SqlTransaction transaccion = oConexion.BeginTransaction();
                try
                {
                    const string qVenta = "SELECT id_sucursal, estado FROM venta WHERE id_venta = @id";
                    SqlCommand cmdVenta = new SqlCommand(qVenta, oConexion, transaccion);
                    cmdVenta.Parameters.AddWithValue("@id", idVenta);

                    int idSucursal = 0;
                    int estado = 0;
                    using (SqlDataReader dr = cmdVenta.ExecuteReader())
                    {
                        if (dr.Read())
                        {
                            idSucursal = Convert.ToInt32(dr["id_sucursal"]);
                            estado = Convert.ToInt32(dr["estado"]);
                        }
                    }

                    // SOLUCIÓN S112: Se usa InvalidOperationException
                    if (estado == 2) throw new InvalidOperationException("La venta ya se encuentra anulada.");

                    const string qUpdate = "UPDATE venta SET estado = 2 WHERE id_venta = @id";
                    SqlCommand cmdUpdate = new SqlCommand(qUpdate, oConexion, transaccion);
                    cmdUpdate.Parameters.AddWithValue("@id", idVenta);
                    cmdUpdate.ExecuteNonQuery();

                    const string qDetalle = "SELECT id_producto, cantidad FROM detalleVenta WHERE id_venta = @id";
                    SqlCommand cmdDetalle = new SqlCommand(qDetalle, oConexion, transaccion);
                    cmdDetalle.Parameters.AddWithValue("@id", idVenta);

                    List<DetalleVenta> detalles = new List<DetalleVenta>();
                    using (SqlDataReader dr = cmdDetalle.ExecuteReader())
                    {
                        while (dr.Read())
                        {
                            detalles.Add(new DetalleVenta()
                            {
                                IdProducto = Convert.ToInt32(dr["id_producto"]),
                                Cantidad = Convert.ToInt32(dr["cantidad"])
                            });
                        }
                    }

                    foreach (var d in detalles)
                    {
                        const string qStock = @"
                            UPDATE stock_sucursal SET cantidad = cantidad + @cant 
                            OUTPUT INSERTED.id_stock_sucursal
                            WHERE id_producto = @idProd AND id_sucursal = @idSuc";

                        SqlCommand cmdStock = new SqlCommand(qStock, oConexion, transaccion);
                        cmdStock.Parameters.AddWithValue("@cant", d.Cantidad);
                        cmdStock.Parameters.AddWithValue("@idProd", d.IdProducto);
                        cmdStock.Parameters.AddWithValue("@idSuc", idSucursal);

                        object objStock = cmdStock.ExecuteScalar();
                        if (objStock != null)
                        {
                            int idStockSucursal = Convert.ToInt32(objStock);

                            const string qMov = @"
                                INSERT INTO movimiento_stock (id_stock_sucursal, cantidad_variacion, tipo_movimiento, id_usuario) 
                                VALUES (@idStock, @cant, 'Anulación', @idUsr)";
                            SqlCommand cmdMov = new SqlCommand(qMov, oConexion, transaccion);
                            cmdMov.Parameters.AddWithValue("@idStock", idStockSucursal);
                            cmdMov.Parameters.AddWithValue("@cant", d.Cantidad);
                            cmdMov.Parameters.AddWithValue("@idUsr", idUsuarioAuditoria);
                            cmdMov.ExecuteNonQuery();
                        }
                    }

                    transaccion.Commit();
                    respuesta = true;
                }
                catch (Exception ex)
                {
                    transaccion.Rollback();
                    Console.WriteLine("Error al anular venta: " + ex.Message);
                }
            }
            return respuesta;
        }
    }

                dgvAuditoria.Rows.Add(
                    item.IdMovimiento,
                    item.Fecha,
                    item.Producto,
                    item.Sucursal,
                    item.TipoMovimiento,
                    item.CantidadVariacion,
                    item.Usuario
                );
            }

            foreach (DataGridViewRow fila in dgvAuditoria.Rows)
            {
                if (int.TryParse(fila.Cells["Cantidad"].Value?.ToString(), out int cantidad))
                {
                    if (cantidad > 0)
                    {
                        fila.Cells["Cantidad"].Style.ForeColor = Color.MediumSeaGreen;
                        fila.Cells["Cantidad"].Style.Font = new Font(dgvAuditoria.Font, FontStyle.Bold);
                        fila.Cells["Cantidad"].Value = "+" + cantidad;
                    }
                    else if (cantidad < 0)
                    {
                        fila.Cells["Cantidad"].Style.ForeColor = EstiloApp.RojoNeon;
                        fila.Cells["Cantidad"].Style.Font = new Font(dgvAuditoria.Font, FontStyle.Bold);
                    }
                }
            }
            dgvAuditoria.ClearSelection();
        }

        private void CargarListaBackupsLocales()
        {
            dgvBackupsLocales.Rows.Clear();
            string ruta = string.IsNullOrWhiteSpace(txtRutaDestino.Text) ? txtRutaDestino.TextButton : txtRutaDestino.Text;

            if (Directory.Exists(ruta))
            {
                DirectoryInfo dir = new DirectoryInfo(ruta);
                foreach (FileInfo archivo in dir.GetFiles("*.bak"))
                {
                    dgvBackupsLocales.Rows.Add(archivo.Name, archivo.FullName, archivo.CreationTime.ToString("dd/MM/yyyy HH:mm:ss"));
                }
            }
            dgvBackupsLocales.ClearSelection();
        }

        private void EjecutarProcesoBackup(bool mostrarMensajes)
        {
            string ruta = string.IsNullOrWhiteSpace(txtRutaDestino.Text) ? txtRutaDestino.TextButton : txtRutaDestino.Text;

            if (!Directory.Exists(ruta))
            {
                try { Directory.CreateDirectory(ruta); }
                catch
                {
                    if (mostrarMensajes)
                        WinForms.MessageBox.Show("La ruta no es válida o no tiene permisos.", "Error", WinForms.MessageBoxButtons.OK, WinForms.MessageBoxIcon.Error);
                    return;
                }
            }

            string nombreArchivo = $"AorusMarket_{DateTime.Now:yyyyMMdd_HHmmss}.bak";
            string rutaCompleta = Path.Combine(ruta, nombreArchivo);

            string mensaje;
            bool exito = _backupNegocio.GenerarRespaldo(rutaCompleta, cmbDestino.SelectedItem.ToString(), out mensaje);

            if (exito)
            {
                CargarListaBackupsLocales();
                if (mostrarMensajes)
                    WinForms.MessageBox.Show($"Respaldo generado con éxito en:\n{rutaCompleta}", "Éxito", WinForms.MessageBoxButtons.OK, WinForms.MessageBoxIcon.Information);
            }
            else
            {
                if (mostrarMensajes)
                    WinForms.MessageBox.Show("Error al generar el backup: " + mensaje, "Error", WinForms.MessageBoxButtons.OK, WinForms.MessageBoxIcon.Error);
            }
        }

        // ==============================================================
        // EJECUCIÓN DE RESTAURACIÓN CON VALIDACIÓN PREVIA
        // ==============================================================
        private void BtnRestaurarBackup_Click(object sender, EventArgs e)
        {
            if (dgvBackupsLocales.CurrentRow == null)
            {
                WinForms.MessageBox.Show("Seleccione un archivo de respaldo local de la tabla para restaurar.", "Aviso", WinForms.MessageBoxButtons.OK, WinForms.MessageBoxIcon.Warning);
                return;
            }

            string archivoSeleccionado = dgvBackupsLocales.CurrentRow.Cells["Ruta"].Value.ToString();

            WinForms.DialogResult alertaCritica = WinForms.MessageBox.Show(
                "¡ATENCIÓN ACCIÓN CRÍTICA!\n\n" +
                "Restaurar la base de datos sobrescribirá toda la información actual por el estado del archivo seleccionado.\n\n" +
                "¿Confirma que posee autorización gerencial para ejecutar esta restauración?",
                "Control Cruzado de Seguridad",
                WinForms.MessageBoxButtons.YesNo,
                WinForms.MessageBoxIcon.Stop);

            if (alertaCritica == WinForms.DialogResult.No) return;

            WinForms.DialogResult confirmacionFinal = WinForms.MessageBox.Show(
                "Esta acción es irreversible y reiniciará el sistema.\n\n¿Está absolutamente seguro?",
                "Última Advertencia",
                WinForms.MessageBoxButtons.YesNo,
                WinForms.MessageBoxIcon.Warning);

            if (confirmacionFinal == WinForms.DialogResult.Yes)
            {
                // Validación Modal de Control Cruzado
                if (!SolicitarAutorizacionCruzada())
                {
                    WinForms.MessageBox.Show("Restauración cancelada. No se obtuvo la autorización cruzada de seguridad.", "Operación Abortada", WinForms.MessageBoxButtons.OK, WinForms.MessageBoxIcon.Warning);
                    return;
                }

                string mensaje;
                bool restaurado = _backupNegocio.RestaurarBaseDatos(archivoSeleccionado, out mensaje);

                if (restaurado)
                {
                    WinForms.MessageBox.Show("Sistema restaurado con éxito. La aplicación se reiniciará.", "Éxito", WinForms.MessageBoxButtons.OK, WinForms.MessageBoxIcon.Information);
                    WinForms.Application.Restart();

        private void EjecutarFiltros()
        {
            int idSucursal = Convert.ToInt32(cmbSucursal.SelectedValue);
            DateTime desde = dtDesde.Value;
            DateTime hasta = dtHasta.Value;

            string mensajeError;

            MetricasDashboard metricas = _dashboardNegocio.ObtenerMetricas(idSucursal, desde, hasta, out mensajeError);

            if (!string.IsNullOrEmpty(mensajeError))
            {
                MessageBox.Show(mensajeError, "Aviso", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            lblTotalVendido.Text = metricas.TotalVendido.ToString("C2");
            lblCantidadVentas.Text = metricas.CantidadVentas.ToString();
            lblProductoTop.Text = string.IsNullOrEmpty(metricas.ProductoMasVendido) ? "-" : metricas.ProductoMasVendido;

            dgvVentasRecientes.Rows.Clear();
            var listaVentas = _dashboardNegocio.ObtenerVentasRecientes(idSucursal, desde, hasta);

            foreach (var v in listaVentas)
            {
                dgvVentasRecientes.Rows.Add(
                    v.IdVenta,
                    v.Fecha.ToString("dd/MM/yyyy HH:mm"),
                    v.Sucursal,
                    v.Cliente,
                    v.Total.ToString("C2"),
                    v.MetodoPago,
                    v.Estado
                );
            }
            dgvVentasRecientes.ClearSelection();
        }

        private void CargarGrilla()
        {
            if (dtpDesde.Value.Date > dtpHasta.Value.Date)
            {
                MessageBox.Show("La fecha 'Desde' no puede ser mayor que 'Hasta'.", "Aviso", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            dgvHistorial.SelectionChanged -= DgvHistorial_SelectionChanged;
            dgvHistorial.Rows.Clear();
            cmbProductosVendidos.Items.Clear();

            int idUsuarioFiltro = Utilidades.SesionActual.IdPerfil == 1 ? 0 : Utilidades.SesionActual.IdUsuario;
            var lista = _ventaNegocio.ListarHistorial(dtpDesde.Value, dtpHasta.Value, idUsuarioFiltro);

            foreach (var item in lista)
            {
                dgvHistorial.Rows.Add(
                    item.IdVenta, item.Fecha, item.Cliente, item.Sucursal,
                    item.Cajero, item.Total, item.MetodoPago, item.Estado
                );
            }

            foreach (DataGridViewRow fila in dgvHistorial.Rows)
            {
                if (fila.Cells["Estado"].Value.ToString() == "Anulada")
                {
                    fila.DefaultCellStyle.ForeColor = Color.IndianRed;
                    fila.DefaultCellStyle.Font = new Font(dgvHistorial.Font, FontStyle.Strikeout);
                }
            }

            dgvHistorial.ClearSelection();
            idVentaSeleccionada = 0;
            dgvHistorial.SelectionChanged += DgvHistorial_SelectionChanged;
        }

        private void BtnBuscar_Click(object sender, EventArgs e)
        {
            CargarGrilla();
        }

        private void DgvHistorial_SelectionChanged(object sender, EventArgs e)
        {
            cmbProductosVendidos.Items.Clear();

            if (dgvHistorial.CurrentRow == null) return;

            var fila = dgvHistorial.CurrentRow;
            idVentaSeleccionada = Convert.ToInt32(fila.Cells["IdVenta"].Value);
            estadoSeleccionado = fila.Cells["Estado"].Value.ToString();

            var detalles = _ventaNegocio.ObtenerDetallesDeVenta(idVentaSeleccionada);

            foreach (var det in detalles)
            {
                string textoCombo = $"{det.Cantidad}x {det.NombreProducto} | Unitario: ${det.PrecioUnitario} | Subtotal: ${det.SubTotal}";
                cmbProductosVendidos.Items.Add(textoCombo);
            }

            if (cmbProductosVendidos.Items.Count > 0)
                cmbProductosVendidos.SelectedIndex = 0;
            else
                cmbProductosVendidos.Items.Add("No hay detalles registrados.");
        }

        private void BtnDescargarFactura_Click(object sender, EventArgs e)
        {
            if (idVentaSeleccionada == 0)
            {
                MessageBox.Show("Seleccione una venta de la grilla para descargar su factura.", "Aviso", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            var fila = dgvHistorial.CurrentRow;
            string fecha = fila.Cells["Fecha"].Value.ToString();
            string cliente = fila.Cells["Cliente"].Value.ToString();
            string cajero = fila.Cells["Cajero"].Value.ToString();
            decimal total = Convert.ToDecimal(fila.Cells["Total"].Value);
            string metodoPago = fila.Cells["MetodoPago"].Value.ToString();

            var detallesBD = _ventaNegocio.ObtenerDetallesDeVenta(idVentaSeleccionada);

            Utilidades.GeneradorFactura.GenerarPDF(idVentaSeleccionada, fecha, cliente, cajero, total, metodoPago, detallesBD, true);
        }

        private void BtnAnular_Click(object sender, EventArgs e)
        {
            if (idVentaSeleccionada == 0)
            {
                MessageBox.Show("Seleccione una venta de la lista para anular.", "Aviso", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            var resp = MessageBox.Show($"¿Está seguro que desea ANULAR el Ticket N° {idVentaSeleccionada}?\n\nEsta acción devolverá los productos al stock permanentemente.",
                "Alerta de Seguridad", MessageBoxButtons.YesNo, MessageBoxIcon.Error);

            if (resp == DialogResult.Yes)
            {
                string mensaje;
                bool resultado = _ventaNegocio.AnularVenta(idVentaSeleccionada, estadoSeleccionado, Utilidades.SesionActual.IdUsuario, out mensaje);

                if (resultado)
                {
                    MessageBox.Show("Venta anulada con éxito. El stock ha sido restituido.", "Operación Exitosa", MessageBoxButtons.OK, MessageBoxIcon.Information);
                    CargarGrilla();
                }
                else
                {
                    MessageBox.Show(mensaje, "Error al anular", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
        }
    }

        private void CargarGrilla(string filtro = "")
        {
            dgvProductos.SelectionChanged -= DgvProductos_SelectionChanged;
            dgvProductos.Rows.Clear();

            var listaProductos = _productoNegocio.Listar();
            if (!string.IsNullOrEmpty(filtro))
                listaProductos = listaProductos.Where(p => p.Nombre.ToLower().Contains(filtro.ToLower())).ToList();

            foreach (var item in listaProductos)
            {
                dgvProductos.Rows.Add(item.IdProducto, item.CodigoBarras, item.Nombre, item.Marca, item.Descripcion, item.UnidadMedida, item.NombreCategoria, item.IdCategoria);
            }
            dgvProductos.ClearSelection();
            dgvProductos.SelectionChanged += DgvProductos_SelectionChanged;
        }

        private void DgvProductos_SelectionChanged(object sender, EventArgs e)
        {
            if (dgvProductos.CurrentRow == null) return;

            idSeleccionado = Convert.ToInt32(fila.Cells["IdProducto"].Value ?? 0);
            EstiloApp.EscribirTexto(txtCodigoBarras, fila.Cells["CodigoBarras"].Value?.ToString());

        private void BtnGuardar_Click(object sender, EventArgs e)
        {
            Producto nuevoProducto = new Producto()
            {
                IdProducto = idSeleccionado,
                CodigoBarras = EstiloApp.LeerTexto(txtCodigoBarras),
                Nombre = EstiloApp.LeerTexto(txtNombre),
                Marca = EstiloApp.LeerTexto(txtMarca),
                Descripcion = EstiloApp.LeerTexto(txtDescripcion),
                UnidadMedida = EstiloApp.LeerTexto(txtUnidadMedida),
                IdCategoria = cmbCategoria.SelectedValue != null ? Convert.ToInt32(cmbCategoria.SelectedValue) : 0
            };

            if (_productoNegocio.Guardar(nuevoProducto, out string mensaje))
            {
                MessageBox.Show("Producto guardado correctamente.", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                LimpiarCampos();
            }
            else
            {
                MessageBox.Show(mensaje, "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void BtnEliminar_Click(object sender, EventArgs e)
        {
            if (idSeleccionado == 0) return;

                }
                else

                    MessageBox.Show(mensaje, "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }

        private void BtnBuscar_Click(object sender, EventArgs e)
        {
            string textoBusqueda = string.IsNullOrWhiteSpace(txtBuscar.TextButton) ? txtBuscar.Text : txtBuscar.TextButton;
            textoBusqueda = textoBusqueda?.Trim();

            if (string.IsNullOrEmpty(textoBusqueda))
            {
                MessageBox.Show("Escriba el código de barras o nombre a buscar.", TituloAviso, MessageBoxButtons.OK, MessageBoxIcon.Information);
                return;
            }

            dgvBusqueda.Rows.Clear();
            var resultados = _ventaNegocio.BuscarProducto(textoBusqueda, SesionActual.IdSucursal);

            if (resultados.Count == 1 && resultados[0].CodigoBarras == textoBusqueda)
            {
                AgregarAlCarritoDirecto(resultados[0], 1);
                EstiloApp.EscribirTexto(txtBuscar, "");
                txtBuscar.Focus();
                return;
            }

            foreach (var r in resultados)
            {
                dgvBusqueda.Rows.Add(r.IdProducto, r.Nombre, r.Precio, r.StockDisponible, "1");
            }
            dgvBusqueda.ClearSelection();
        }

        private void AgregarAlCarritoDirecto(BusquedaProducto producto, int cantidadAInsertar)
        {
            int cantidadYaEnCarrito = 0;
            foreach (DataGridViewRow row in dgvCarrito.Rows)
            {
                if (Convert.ToInt32(row.Cells[ColIdProducto].Value) == producto.IdProducto)
                {
                    cantidadYaEnCarrito += Convert.ToInt32(row.Cells[ColCantidad].Value);
                }
            }

            int cantidadTotalDeseada = cantidadAInsertar + cantidadYaEnCarrito;

            if (cantidadTotalDeseada > producto.StockDisponible)
            {
                MessageBox.Show($"No hay stock suficiente. Tienes {producto.StockDisponible} disponibles.", "Stock Insuficiente", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            if (!_stockNegocio.HayStockSuficiente(producto.IdProducto, SesionActual.IdSucursal, cantidadTotalDeseada))
            {
                MessageBox.Show("El stock en la base de datos ya no es suficiente.", "Error de Stock", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return;
            }

            bool existe = false;
            foreach (DataGridViewRow row in dgvCarrito.Rows)
            {
                if (Convert.ToInt32(row.Cells[ColIdProducto].Value) == producto.IdProducto)
                {
                    row.Cells[ColCantidad].Value = cantidadTotalDeseada;
                    row.Cells[ColSubtotal].Value = cantidadTotalDeseada * producto.Precio;
                    existe = true;
                    break;
                }
            }

            if (!existe)
            {
                dgvCarrito.Rows.Add(producto.IdProducto, producto.Nombre, cantidadAInsertar, producto.Precio, producto.Precio * cantidadAInsertar);
            }

            RecalcularTotal();
        }

            {

                Nombre = fila.Cells[ColProducto].Value.ToString(),

                StockDisponible = Convert.ToInt32(fila.Cells["Disponible"].Value)

            AgregarAlCarritoDirecto(bp, cantIngresada);
            fila.Cells[ColCantidad].Value = "";
        }

        // SOLUCIÓN A COMPLEJIDAD COGNITIVA (S3776): Se extrajo la validación inicial
        private bool ValidarPreVenta(out string metodoSeleccionado)
        {
            metodoSeleccionado = string.Empty;
            if (dgvCarrito.Rows.Count == 0)
            {
                MessageBox.Show("El carrito está vacío.", TituloAviso, MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return false;
            }

            if (cmbMetodoPago.SelectedIndex == -1)
            {
                MessageBox.Show("Seleccione un método de pago.", TituloAviso, MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return false;
            }

            metodoSeleccionado = cmbMetodoPago.SelectedItem.ToString();
            return true;
        }

        private bool ValidarEfectivoOConfirmar(string metodoSeleccionado)
        {
            if (metodoSeleccionado == "Efectivo")
            {
                decimal montoAbonado = PedirMontoEfectivo(totalAcumulado);
                if (montoAbonado == -1) return false;

                if (montoAbonado < totalAcumulado)
                {
                    MessageBox.Show("El monto ingresado es menor al total de la venta.", "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
                    return false;
                }

                decimal vuelto = montoAbonado - totalAcumulado;
                return MessageBox.Show($"Total a cobrar: {totalAcumulado:C2}\nAbona con: {montoAbonado:C2}\n\nSU VUELTO ES: {vuelto:C2}\n\n¿Confirmar venta?", "Verificación", MessageBoxButtons.YesNo, MessageBoxIcon.Question) == DialogResult.Yes;
            }

            return MessageBox.Show($"¿Confirmar venta por un total de {totalAcumulado:C2}?", "Validación", MessageBoxButtons.YesNo, MessageBoxIcon.Question) == DialogResult.Yes;
        }

        private void BtnConfirmar_Click(object sender, EventArgs e)
        {
            if (!ValidarPreVenta(out string metodoSeleccionado)) return;
            if (!ValidarEfectivoOConfirmar(metodoSeleccionado)) return;

            if (totalAcumulado > 50000)
            {
                if (MessageBox.Show("Esta venta supera el límite de $50,000. ¿Desea continuar de todas formas?", "Autorización Requerida", MessageBoxButtons.YesNo, MessageBoxIcon.Warning) == DialogResult.No) return;
            }

            ProcesarCierreVenta(metodoSeleccionado);
        }

        private void ProcesarCierreVenta(string metodoSeleccionado)
        {
            Venta nuevaVenta = new Venta()
            {
                Total = totalAcumulado,
                MetodoPago = metodoSeleccionado,
                IdUsuario = SesionActual.IdUsuario,
                IdSucursal = SesionActual.IdSucursal,
                IdCliente = Convert.ToInt32(cmbCliente.SelectedValue) == 0 ? (int?)null : Convert.ToInt32(cmbCliente.SelectedValue)
            };

            foreach (DataGridViewRow fila in dgvCarrito.Rows)
            {
                nuevaVenta.Detalles.Add(new DetalleVenta()
                {
                    IdProducto = Convert.ToInt32(fila.Cells[ColIdProducto].Value),
                    Cantidad = Convert.ToInt32(fila.Cells[ColCantidad].Value),
                    PrecioUnitario = Convert.ToDecimal(fila.Cells[ColPrecioUnit].Value)
                });
            }

            if (_ventaNegocio.ConfirmarVenta(nuevaVenta, out string mensaje))
            {
                MessageBox.Show("¡Venta procesada con éxito!", "AorusMarket", MessageBoxButtons.OK, MessageBoxIcon.Information);

                if (MessageBox.Show("¿Desea generar y descargar la Factura en PDF ahora?", "Factura", MessageBoxButtons.YesNo, MessageBoxIcon.Question) == DialogResult.Yes)
                {
                    List<DetalleHistorial> detallesPdf = new List<DetalleHistorial>();
                    foreach (DataGridViewRow row in dgvCarrito.Rows)
                    {
                        detallesPdf.Add(new DetalleHistorial()
                        {
                            NombreProducto = row.Cells[ColProducto].Value.ToString(),
                            Cantidad = Convert.ToInt32(row.Cells[ColCantidad].Value),
                            PrecioUnitario = Convert.ToDecimal(row.Cells[ColPrecioUnit].Value),
                            SubTotal = Convert.ToDecimal(row.Cells[ColSubtotal].Value)
                        });
                    }

                    string nombreCliente = cmbCliente.Text.Contains("-") ? cmbCliente.Text.Split('-')[1].Trim() : "Consumidor Final";
                    GeneradorFactura.GenerarPDF(nuevaVenta.IdVenta, DateTime.Now.ToString("dd/MM/yyyy HH:mm"), nombreCliente, SesionActual.NombreCompleto, nuevaVenta.Total, nuevaVenta.MetodoPago, detallesPdf, true);
                }
                LimpiarVenta();
            }
            else
            {
                MessageBox.Show(mensaje, "Error en Venta", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void LimpiarVenta()
        {
            dgvCarrito.Rows.Clear();
            dgvBusqueda.Rows.Clear();
            cmbCliente.SelectedIndex = 0;
            cmbMetodoPago.SelectedIndex = -1;
            EstiloApp.EscribirTexto(txtBuscar, "");
            RecalcularTotal();
        }
    }

        private void BtnGenerar_Click(object sender, EventArgs e)

            switch (cmbTipoReporte.SelectedIndex)

                case 0: dgvReporte.DataSource = _reporteNegocio.RecaudacionPorMedioPago(desde, hasta); break;
                case 1: dgvReporte.DataSource = _reporteNegocio.EvolucionVentas(desde, hasta); break;

        private void DarFormatoColumnas()
        {
            if (dgvReporte.Columns.Count == 0) return;

            foreach (DataGridViewColumn col in dgvReporte.Columns)
            {
                if (col.Name.Contains("Total") || col.Name.Contains("Recaudado") || col.Name.Contains("Vendido"))
                {
                    col.DefaultCellStyle.Format = "C2";
                    col.DefaultCellStyle.Alignment = DataGridViewContentAlignment.MiddleRight;
                }
                else if (col.Name.Contains("Cantidad"))
                {
                    col.DefaultCellStyle.Alignment = DataGridViewContentAlignment.MiddleCenter;
                }
                else
                {
                    col.DefaultCellStyle.Alignment = DataGridViewContentAlignment.MiddleLeft;
                }
                col.HeaderText = System.Text.RegularExpressions.Regex.Replace(col.HeaderText, "([A-Z])", " $1").Trim();
            }
            dgvReporte.ClearSelection();
        }

        private void BtnGenerarGrafico_Click(object sender, EventArgs e)
        {
            if (dgvReporte.Rows.Count == 0)
            {
                MessageBox.Show("Primero debe generar una tabla de datos para poder crear el gráfico.", "Aviso", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            SaveFileDialog sfd = new SaveFileDialog();
            sfd.Filter = "Archivo PDF (*.pdf)|*.pdf";
            sfd.FileName = $"Grafico_Aorus_{DateTime.Now:ddMMyyyy}.pdf";

            if (sfd.ShowDialog() == DialogResult.OK)
            {
                try
                {
                    Chart graficoTemp = new Chart();
                    graficoTemp.Size = new Size(900, 500);
                    graficoTemp.BackColor = Color.White;

                    ChartArea area = new ChartArea();
                    area.AxisX.MajorGrid.LineColor = Color.LightGray;
                    area.AxisY.MajorGrid.LineColor = Color.LightGray;
                    area.AxisX.LabelStyle.Font = new System.Drawing.Font("Segoe UI", 10F);
                    area.AxisY.LabelStyle.Font = new System.Drawing.Font("Segoe UI", 10F);
                    graficoTemp.ChartAreas.Add(area);

                    Legend leyenda = new Legend();
                    leyenda.Font = new System.Drawing.Font("Segoe UI", 11F);
                    graficoTemp.Legends.Add(leyenda);

                    Series serie = new Series("Datos");
                    serie.IsValueShownAsLabel = true;
                    serie.Font = new System.Drawing.Font("Segoe UI", 11F, FontStyle.Bold);
                    serie.Palette = ChartColorPalette.Pastel;

                    int indexReporte = cmbTipoReporte.SelectedIndex;
                    if (indexReporte == 0) serie.ChartType = SeriesChartType.Pie;
                    else if (indexReporte == 1) { serie.ChartType = SeriesChartType.Line; serie.BorderWidth = 4; serie.Color = Color.Crimson; }
                    else if (indexReporte == 2) serie.ChartType = SeriesChartType.Column;
                    else if (indexReporte == 3) serie.ChartType = SeriesChartType.Bar;
                    else if (indexReporte == 4) serie.ChartType = SeriesChartType.Doughnut;

                    foreach (DataGridViewRow row in dgvReporte.Rows)
                    {
                        string nombreEjeX = row.Cells[0].Value.ToString();
                        decimal valorEjeY = Convert.ToDecimal(row.Cells[row.Cells.Count - 1].Value);
                        serie.Points.AddXY(nombreEjeX, valorEjeY);
                    }

                    graficoTemp.Series.Add(serie);
                    graficoTemp.Update();

                    using (MemoryStream ms = new MemoryStream())
                    {
                        graficoTemp.SaveImage(ms, ChartImageFormat.Png);
                        iTextSharp.text.Image imagenGrafico = iTextSharp.text.Image.GetInstance(ms.GetBuffer());
                        imagenGrafico.Alignment = iTextSharp.text.Element.ALIGN_CENTER;
                        imagenGrafico.ScaleToFit(520f, 400f);

                        Document doc = new Document(PageSize.A4, 30, 30, 30, 30);
                        PdfWriter.GetInstance(doc, new FileStream(sfd.FileName, FileMode.Create));
                        doc.Open();

                        string textoReporte = cmbTipoReporte.Text.Substring(3).ToUpper();
                        iTextSharp.text.Font fontTitulo = FontFactory.GetFont(FontFactory.HELVETICA_BOLD, 18);
                        Paragraph titulo = new Paragraph($"AORUS MARKET - GRÁFICO DE {textoReporte}", fontTitulo);
                        titulo.Alignment = Element.ALIGN_CENTER;
                        doc.Add(titulo);

                        Paragraph subtitulo = new Paragraph($"Período: {dtpDesde.Value:dd/MM/yyyy} al {dtpHasta.Value:dd/MM/yyyy}", FontFactory.GetFont(FontFactory.HELVETICA, 12));
                        subtitulo.Alignment = Element.ALIGN_CENTER;
                        doc.Add(subtitulo);
                        doc.Add(new Chunk("\n\n"));

                        doc.Add(imagenGrafico);
                        doc.Close();
                    }

                    System.Diagnostics.Process.Start(sfd.FileName);
                }
                catch (Exception ex)
                {
                    MessageBox.Show("Error al generar el gráfico: " + ex.Message, "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }

        private void CargarGrilla()
        {
            dgvSucursales.SelectionChanged -= DgvSucursales_SelectionChanged;

            dgvSucursales.Rows.Clear();
            var listaSucursales = _sucursalNegocio.Listar();

            foreach (var item in listaSucursales)
            {
                dgvSucursales.Rows.Add(item.IdSucursal, item.Nombre, item.Direccion, item.Telefono);
            }
            dgvSucursales.ClearSelection();

            dgvSucursales.SelectionChanged += DgvSucursales_SelectionChanged;
        }

        private void DgvSucursales_SelectionChanged(object sender, EventArgs e)
        {
            if (dgvSucursales.CurrentRow == null) return;
            var fila = dgvSucursales.CurrentRow;

            idSeleccionado = Convert.ToInt32(fila.Cells["IdSucursal"].Value ?? 0);

            string valNombre = fila.Cells["Nombre"].Value?.ToString() ?? "";
            string valDireccion = fila.Cells["Direccion"].Value?.ToString() ?? "";
            string valTelefono = fila.Cells["Telefono"].Value?.ToString() ?? "";

            txtNombre.Text = valNombre;
            txtNombre.TextButton = valNombre;

            txtDireccion.Text = valDireccion;
            txtDireccion.TextButton = valDireccion;

            txtTelefono.Text = valTelefono;
            txtTelefono.TextButton = valTelefono;
        }

        private void LimpiarCampos()
        {
            idSeleccionado = 0;

            txtNombre.Text = "";
            txtNombre.TextButton = "";

            txtDireccion.Text = "";
            txtDireccion.TextButton = "";

            txtTelefono.Text = "";
            txtTelefono.TextButton = "";

            dgvSucursales.ClearSelection();
        }

        private void BtnGuardar_Click(object sender, EventArgs e)
        {
            string nombreIngresado = string.IsNullOrWhiteSpace(txtNombre.Text) ? txtNombre.TextButton : txtNombre.Text;
            string direccionIngresada = string.IsNullOrWhiteSpace(txtDireccion.Text) ? txtDireccion.TextButton : txtDireccion.Text;
            string telefonoIngresado = string.IsNullOrWhiteSpace(txtTelefono.Text) ? txtTelefono.TextButton : txtTelefono.Text;

            Sucursal nuevaSucursal = new Sucursal()
            {
                IdSucursal = idSeleccionado,
                Nombre = nombreIngresado?.Trim(),
                Direccion = direccionIngresada?.Trim(),
                Telefono = telefonoIngresado?.Trim()
            };

            string mensaje;
            bool resultado = _sucursalNegocio.Guardar(nuevaSucursal, out mensaje);

            if (resultado)
            {
                WinForms.MessageBox.Show("Sucursal guardada correctamente.", "Éxito", WinForms.MessageBoxButtons.OK, WinForms.MessageBoxIcon.Information);
                LimpiarCampos();
                CargarGrilla();
            }
            else
            {
                WinForms.MessageBox.Show(mensaje, "Error al guardar", WinForms.MessageBoxButtons.OK, WinForms.MessageBoxIcon.Error);
            }
        }

        private void BtnEliminar_Click(object sender, EventArgs e)
        {
            if (idSeleccionado == 0)
            {
                WinForms.MessageBox.Show("Seleccione una sucursal de la tabla", "Aviso",
                    WinForms.MessageBoxButtons.OK, WinForms.MessageBoxIcon.Warning);
                return;
            }

            var resp = WinForms.MessageBox.Show("¿Seguro que desea eliminar esta sucursal?", "Confirmar Eliminación",
                WinForms.MessageBoxButtons.YesNo, WinForms.MessageBoxIcon.Exclamation);

            if (resp == WinForms.DialogResult.Yes)
            {
                string mensaje;
                bool resultado = _sucursalNegocio.Eliminar(idSeleccionado, out mensaje);

                if (resultado)
                {
                    WinForms.MessageBox.Show("Sucursal eliminada.", "Éxito", WinForms.MessageBoxButtons.OK, WinForms.MessageBoxIcon.Information);
                    LimpiarCampos();
                    CargarGrilla();
                }
                else
                {
                    WinForms.MessageBox.Show(mensaje, "Error", WinForms.MessageBoxButtons.OK, WinForms.MessageBoxIcon.Error);
                }
            }
        }
    }

        private void CargarCombos()
        {
            cmbPerfil.DataSource = _usuarioNegocio.ObtenerPerfiles();
            cmbPerfil.DisplayMember = "Nombre";
            cmbPerfil.ValueMember = "Id";
            cmbPerfil.SelectedIndex = -1;

            cmbSucursal.DataSource = _usuarioNegocio.ObtenerSucursales();
            cmbSucursal.DisplayMember = "Nombre";
            cmbSucursal.ValueMember = "Id";
            cmbSucursal.SelectedIndex = -1;
        }

        private void CargarGrilla(string filtro = "")
        {
            dgvUsuarios.SelectionChanged -= DgvUsuarios_SelectionChanged;
            dgvUsuarios.Rows.Clear();

            var listaUsuarios = _usuarioNegocio.Listar(filtro);
            foreach (var item in listaUsuarios)
            {
                dgvUsuarios.Rows.Add(item.IdUsuario, item.Dni, item.Nombre, item.Apellido, item.Email, item.Telefono, item.Calle, item.Altura, item.FechaNacimiento.HasValue ? item.FechaNacimiento.Value.ToString("dd/MM/yyyy") : "", item.NombrePerfil, item.NombreSucursal, item.IdPerfil, item.IdSucursal, item.IdDireccion);
            }

            dgvUsuarios.ClearSelection();
            dgvUsuarios.SelectionChanged += DgvUsuarios_SelectionChanged;
        }

        private void DgvUsuarios_SelectionChanged(object sender, EventArgs e)
        {
            if (dgvUsuarios.CurrentRow == null) return;

            var fila = dgvUsuarios.CurrentRow;
            idSeleccionado = Convert.ToInt32(fila.Cells["IdUsuario"].Value ?? 0);
            idDireccionSeleccionada = Convert.ToInt32(fila.Cells["IdDireccion"].Value ?? 0);

            EstiloApp.EscribirTexto(txtDni, fila.Cells["Dni"].Value?.ToString());
            EstiloApp.EscribirTexto(txtNombre, fila.Cells["Nombre"].Value?.ToString());
            EstiloApp.EscribirTexto(txtApellido, fila.Cells["Apellido"].Value?.ToString());
            EstiloApp.EscribirTexto(txtEmail, fila.Cells["Email"].Value?.ToString());
            EstiloApp.EscribirTexto(txtTelefono, fila.Cells["Telefono"].Value?.ToString());
            EstiloApp.EscribirTexto(txtCalle, fila.Cells["Calle"].Value?.ToString());
            EstiloApp.EscribirTexto(txtAltura, fila.Cells["Altura"].Value?.ToString());
            EstiloApp.EscribirTexto(txtPassword, ""); // Siempre limpiar pass al seleccionar

            if (DateTime.TryParse(fila.Cells["FechaNacimiento"].Value?.ToString(), out DateTime fechaNac))
                dtFechaNacimiento.Value = fechaNac;
            else
                dtFechaNacimiento.Value = DateTime.Now;

            cmbPerfil.SelectedValue = Convert.ToInt32(fila.Cells["IdPerfil"].Value ?? 0);
            cmbSucursal.SelectedValue = Convert.ToInt32(fila.Cells["IdSucursal"].Value ?? 0);
        }

        private void LimpiarCampos()
        {
            idSeleccionado = 0;
            idDireccionSeleccionada = 0;
            EstiloApp.EscribirTexto(txtDni, "");
            EstiloApp.EscribirTexto(txtNombre, "");
            EstiloApp.EscribirTexto(txtApellido, "");
            EstiloApp.EscribirTexto(txtEmail, "");
            EstiloApp.EscribirTexto(txtTelefono, "");
            EstiloApp.EscribirTexto(txtCalle, "");
            EstiloApp.EscribirTexto(txtAltura, "");
            EstiloApp.EscribirTexto(txtPassword, "");
            EstiloApp.EscribirTexto(txtBuscarDni, "");
            dtFechaNacimiento.Value = DateTime.Now;
            cmbPerfil.SelectedIndex = -1;
            cmbSucursal.SelectedIndex = -1;
            dgvUsuarios.ClearSelection();
            CargarGrilla();
        }

        private void BtnGuardar_Click(object sender, EventArgs e)
        {
            Usuario nuevoUsuario = new Usuario()
            {
                IdUsuario = idSeleccionado,
                IdDireccion = idDireccionSeleccionada,
                Dni = EstiloApp.LeerTexto(txtDni),
                Nombre = EstiloApp.LeerTexto(txtNombre),
                Apellido = EstiloApp.LeerTexto(txtApellido),
                Email = EstiloApp.LeerTexto(txtEmail),
                Password = EstiloApp.LeerTexto(txtPassword),
                Telefono = EstiloApp.LeerTexto(txtTelefono),
                Calle = EstiloApp.LeerTexto(txtCalle),
                Altura = EstiloApp.LeerTexto(txtAltura),
                FechaNacimiento = dtFechaNacimiento.Value.Date,
                IdPerfil = cmbPerfil.SelectedValue != null ? Convert.ToInt32(cmbPerfil.SelectedValue) : 0,
                IdSucursal = cmbSucursal.SelectedValue != null ? Convert.ToInt32(cmbSucursal.SelectedValue) : 0
            };

            if (_usuarioNegocio.Guardar(nuevoUsuario, out string mensaje))
            {
                MessageBox.Show("Usuario guardado correctamente.", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                LimpiarCampos();
            }
            else
            {
                MessageBox.Show(mensaje, "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void BtnEliminar_Click(object sender, EventArgs e)
        {
            if (idSeleccionado == 0) return;

            if (MessageBox.Show("¿Seguro que desea eliminar este usuario?", "Confirmar", MessageBoxButtons.YesNo, MessageBoxIcon.Exclamation) == DialogResult.Yes)
            {
                if (_usuarioNegocio.Eliminar(idSeleccionado, out string mensaje))
                    LimpiarCampos();
                else
                    MessageBox.Show(mensaje, "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }
    }

        private CyberButton CrearBotonMenu(string texto, int yPos)
        {
            Color colorGrisModerno = Color.LightGray;

            CyberButton btn = new CyberButton
            {
                TextButton = texto,
                Tag = texto,
                Location = new Point(5, yPos),
                Size = new Size(240, 42),
                Alpha = 20,
                Rounding = true,
                RoundingInt = 8,
                Font = new Font("Segoe UI Emoji", 10.5F, FontStyle.Bold),

                ColorBackground_Pen = Color.Transparent,

            btn.MouseEnter += (s, e) =>
            {
                if (botonActivo != btn)
                {
                    btn.ColorBackground = Color.FromArgb(40, 40, 40);
                    btn.ColorBackground_Pen = Color.FromArgb(40, 40, 40);
                    btn.ForeColor = Color.White;

                    btn.ColorBackground_Pen = Color.Transparent;

        private void TimerIndicador_Tick(object sender, EventArgs e)
        {
            if (botonActivo == null) return;

            int distancia = botonActivo.Top - pnlIndicador.Top;

            if (Math.Abs(distancia) <= 1)
            {
                pnlIndicador.Top = botonActivo.Top;
                timerIndicador.Stop();
            }
            else
            {
                pnlIndicador.Top += distancia / 3;
            }
        }

        // SOLUCIÓN CARRITO: Método modificado para usar caché
        private void AbrirFormularioEnPanel<T>() where T : Form, new()
        {
            if (formularioActivo != null)
            {
                formularioActivo.Hide(); // Ocultamos en lugar de destruir
            }

            Type tipo = typeof(T);

            // Si el formulario nunca se abrió, lo creamos y lo guardamos
            if (!formulariosCacheados.ContainsKey(tipo))
            {
                T nuevoFormulario = new T();
                nuevoFormulario.TopLevel = false;
                nuevoFormulario.FormBorderStyle = FormBorderStyle.None;
                nuevoFormulario.Dock = DockStyle.Fill;
                nuevoFormulario.BackColor = Color.FromArgb(18, 18, 18);
                
                panelContenedor.Controls.Add(nuevoFormulario);
                formulariosCacheados.Add(tipo, nuevoFormulario);
            }

            // Llamamos al formulario desde la caché
            formularioActivo = formulariosCacheados[tipo];
            formularioActivo.BringToFront();
            formularioActivo.Show();
        }

        private void AplicarPermisosPorPerfil()
        {
            CyberButton[] todosLosBotones = {
                btnPuntoVenta, btnClientes, btnStock, btnProductos,
                btnCategorias, btnUsuarios, btnSucursales, btnHistorial,
                btnAuditoria, btnBackup, btnDashboard, btnReportes
            };

            foreach (var btn in todosLosBotones)
            {
                if (btn != null) btn.Visible = false;
            }

            List<CyberButton> menuDelUsuario = new List<CyberButton>();

            if (SesionActual.IdPerfil == 1)
            {
                menuDelUsuario.Add(btnDashboard);
                menuDelUsuario.Add(btnUsuarios);
                menuDelUsuario.Add(btnSucursales);
                menuDelUsuario.Add(btnAuditoria);
                menuDelUsuario.Add(btnBackup);
                menuDelUsuario.Add(btnClientes);
                menuDelUsuario.Add(btnHistorial);
                menuDelUsuario.Add(btnStock);
                menuDelUsuario.Add(btnProductos);
                menuDelUsuario.Add(btnCategorias);
                menuDelUsuario.Add(btnReportes);

                ResaltarBotonActivo(btnDashboard);
                AbrirFormularioEnPanel<FrmDashboard>();
            }
            else if (SesionActual.IdPerfil == 2)
            {
                menuDelUsuario.Add(btnPuntoVenta);
                menuDelUsuario.Add(btnClientes);
                menuDelUsuario.Add(btnHistorial); 

                ResaltarBotonActivo(btnPuntoVenta);
                AbrirFormularioEnPanel<FrmPuntoVenta>();
            }
            else if (SesionActual.IdPerfil == 3)
            {
                menuDelUsuario.Add(btnStock);
                menuDelUsuario.Add(btnProductos);
                menuDelUsuario.Add(btnCategorias);

                ResaltarBotonActivo(btnStock);
                AbrirFormularioEnPanel<FrmStock>();
            }

            int yPos = 110;
            int btnHeight = 45;

            foreach (var btn in menuDelUsuario)
            {
                btn.Visible = true;
                btn.Location = new Point(5, yPos);
                yPos += btnHeight;
            }

            if (botonActivo != null)
            {
                pnlIndicador.Top = botonActivo.Top;
            }
        }

        private void BtnCerrarSesion_Click(object sender, EventArgs e)
        {
            var resp = MessageBox.Show("¿Seguro que desea cerrar sesión?", "Confirmar",
                MessageBoxButtons.YesNo, MessageBoxIcon.Question);

            if (resp == DialogResult.Yes)
            {
                // Limpiamos la caché al cerrar sesión para que el próximo usuario entre con todo en blanco
                foreach (var form in formulariosCacheados.Values)
                {
                    form.Close();
                }
                formulariosCacheados.Clear();

                SesionActual.CerrarSesion();
                this.Hide();
                FrmLogin login = new FrmLogin();
                login.Show();
            }
        }

        private void ActivarDoubleBuffering(Control control)
        {
            PropertyInfo prop = typeof(Control).GetProperty("DoubleBuffered",
                BindingFlags.Instance | BindingFlags.NonPublic);

            if (prop != null)
            {
                prop.SetValue(control, true, null);
            }
        }

    public class BackupNegocio
    {
        public bool GenerarRespaldo(string rutaCompleta, string tipoDestino, out string mensaje)
        {
            mensaje = string.Empty;
            try
            {
                using (SqlConnection oConexion = Conexion.ObtenerConexion())
                {
                    oConexion.Open();
                    string nombreBd = oConexion.Database;

                    string query = $"BACKUP DATABASE [{nombreBd}] TO DISK = @ruta WITH FORMAT, MEDIANAME = 'AorusBackups', NAME = 'Backup Completo AorusMarket';";

                    using (SqlCommand cmd = new SqlCommand(query, oConexion))
                    {
                        cmd.Parameters.AddWithValue("@ruta", rutaCompleta);
                        cmd.CommandTimeout = 300;
                        cmd.ExecuteNonQuery();
                    }
                }

                // 1. Conserva solo los 5 más recientes
                string carpeta = Path.GetDirectoryName(rutaCompleta);
                PurgarBackupsAntiguos(carpeta, 5);

                // 2. Si se eligió Google Drive, sube el archivo
                if (tipoDestino.Contains("Google Drive"))
                {
                    SubirAGoogleDrive(rutaCompleta, out string msjDrive);
                    if (!string.IsNullOrEmpty(msjDrive))
                    {
                        mensaje = "Backup local creado. Aviso Drive: " + msjDrive;
                    }
                }

                return true;
            }
            catch (Exception ex)
            {
                mensaje = "Error al generar el respaldo: " + ex.Message;
                return false;
            }
        }

        public void PurgarBackupsAntiguos(string carpeta, int cantidadMaxima)
        {
            try
            {
                if (!Directory.Exists(carpeta)) return;

                DirectoryInfo info = new DirectoryInfo(carpeta);
                var archivos = info.GetFiles("AorusMarket_*.bak")
                                   .OrderByDescending(f => f.CreationTime)
                                   .ToList();

                if (archivos.Count > cantidadMaxima)
                {
                    var aEliminar = archivos.Skip(cantidadMaxima);
                    foreach (var archivo in aEliminar)
                    {
                        try { archivo.Delete(); } catch { }
                    }
                }
            }
            catch { }
        }

        public bool SubirAGoogleDrive(string rutaArchivoBak, out string mensaje)
        {
            mensaje = string.Empty;
            try
            {
                string credencialesPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "credentials.json");

                if (!File.Exists(credencialesPath))
                {
                    mensaje = "Falta el archivo credentials.json en la carpeta del sistema.";
                    return false;
                }

                UserCredential credential;
                using (var stream = new FileStream(credencialesPath, FileMode.Open, FileAccess.Read))
                {
                    string tokenPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "DriveToken");

                    credential = GoogleWebAuthorizationBroker.AuthorizeAsync(
                        GoogleClientSecrets.FromStream(stream).Secrets,
                        new[] { DriveService.Scope.DriveFile },
                        "user",
                        CancellationToken.None,
                        new FileDataStore(tokenPath, true)).Result;
                }

                var service = new DriveService(new BaseClientService.Initializer()
                {
                    HttpClientInitializer = credential,
                    ApplicationName = "AorusMarket"
                });

                var fileMetadata = new Google.Apis.Drive.v3.Data.File()
                {
                    Name = Path.GetFileName(rutaArchivoBak)
                };

                FilesResource.CreateMediaUpload request;
                using (var uploadStream = new FileStream(rutaArchivoBak, FileMode.Open))
                {
                    request = service.Files.Create(fileMetadata, uploadStream, "application/octet-stream");
                    request.Upload();
                }

                return true;
            }
            catch (Exception ex)
            {
                mensaje = "Error al subir a Google Drive: " + ex.Message;
                return false;
            }
        }

        public bool RestaurarBaseDatos(string archivoBak, out string mensaje)
        {
            mensaje = string.Empty;

            if (!File.Exists(archivoBak))
            {
                mensaje = "El archivo de respaldo seleccionado no existe.";
                return false;
            }

            try
            {
                SqlConnectionStringBuilder builder = new SqlConnectionStringBuilder(Conexion.ObtenerConexion().ConnectionString);
                string nombreDbActual = builder.InitialCatalog;
                builder.InitialCatalog = "master";

                using (SqlConnection oConexionMaster = new SqlConnection(builder.ConnectionString))
                {
                    oConexionMaster.Open();

                    string querySingleUser = $"ALTER DATABASE [{nombreDbActual}] SET SINGLE_USER WITH ROLLBACK IMMEDIATE;";
                    using (SqlCommand cmd1 = new SqlCommand(querySingleUser, oConexionMaster))
                    {
                        cmd1.ExecuteNonQuery();
                    }

                    string queryRestore = $"RESTORE DATABASE [{nombreDbActual}] FROM DISK = @ruta WITH REPLACE;";
                    using (SqlCommand cmd2 = new SqlCommand(queryRestore, oConexionMaster))
                    {
                        cmd2.Parameters.AddWithValue("@ruta", archivoBak);
                        cmd2.CommandTimeout = 300;
                        cmd2.ExecuteNonQuery();
                    }

                    string queryMultiUser = $"ALTER DATABASE [{nombreDbActual}] SET MULTI_USER;";
                    using (SqlCommand cmd3 = new SqlCommand(queryMultiUser, oConexionMaster))
                    {
                        cmd3.ExecuteNonQuery();
                    }
                }

                return true;
            }
            catch (Exception ex)
            {
                mensaje = "Error al restaurar: " + ex.Message;
                return false;
            }
        }
    }

namespace AorusMarket.Negocio
{
    public class CategoriaNegocio
    {
        private CategoriaDatos _categoriaDatos = new CategoriaDatos();

        public List<Categoria> Listar()
        {
            return _categoriaDatos.Listar();
        }

        public bool Guardar(Categoria obj, out string mensaje)
        {
            mensaje = string.Empty;

            if (string.IsNullOrWhiteSpace(obj.Nombre))
            {
                mensaje = "El nombre de la categoría no puede estar vacío.";
                return false;
            }

            // Validación de duplicados
            var listaExistente = _categoriaDatos.Listar();
            if (listaExistente.Any(c => c.Nombre.Trim().ToLower() == obj.Nombre.Trim().ToLower() && c.IdCategoria != obj.IdCategoria))
            {
                mensaje = "Ya existe una categoría registrada con ese nombre.";
                return false;
            }

            if (obj.IdCategoria == 0)
            {
                bool resultado = _categoriaDatos.Insertar(obj);
                if (!resultado) mensaje = "No se pudo insertar la categoría.";
                return resultado;
            }
            else
            {
                bool resultado = _categoriaDatos.Editar(obj);
                if (!resultado) mensaje = "No se pudo actualizar la categoría.";
                return resultado;
            }
        }

        public bool Eliminar(int id, out string mensaje)
        {
            mensaje = string.Empty;
            bool resultado = _categoriaDatos.Eliminar(id);
            if (!resultado)
            {
                mensaje = "Ocurrió un error al intentar eliminar la categoría.";
            }
            return resultado;
        }
    }

    public class ClienteNegocio
    {
        private ClienteDatos _clienteDatos = new ClienteDatos();
        public List<Cliente> Listar(string filtroDni = "") => _clienteDatos.Listar(filtroDni);

        public bool Guardar(Cliente obj, out string mensaje)
        {
            mensaje = string.Empty;

            if (string.IsNullOrWhiteSpace(obj.Dni) || !Regex.IsMatch(obj.Dni, @"^\d{7,8}$"))
            { mensaje = "El DNI es obligatorio y debe contener entre 7 y 8 números."; return false; }

            if (string.IsNullOrWhiteSpace(obj.CuilCuit) || !Regex.IsMatch(obj.CuilCuit, @"^\d{11}$"))
            { mensaje = "El CUIL/CUIT es obligatorio y debe tener exactamente 11 números (sin guiones)."; return false; }

            if (string.IsNullOrWhiteSpace(obj.Nombre) || obj.Nombre.Trim().Length < 2)
            { mensaje = "El Nombre es obligatorio y debe tener al menos 2 letras."; return false; }

            if (string.IsNullOrWhiteSpace(obj.Apellido) || obj.Apellido.Trim().Length < 2)
            { mensaje = "El Apellido es obligatorio y debe tener al menos 2 letras."; return false; }

            if (string.IsNullOrWhiteSpace(obj.Email) || !Regex.IsMatch(obj.Email, @"^[^@\s]+@[^@\s]+\.[^@\s]+$"))
            { mensaje = "Debe ingresar un Email válido (ejemplo@dominio.com)."; return false; }

            if (string.IsNullOrWhiteSpace(obj.Telefono) || !Regex.IsMatch(obj.Telefono, @"^\d{6,15}$"))
            { mensaje = "El Teléfono es obligatorio y debe contener solo números."; return false; }

            if (string.IsNullOrWhiteSpace(obj.Calle))
            { mensaje = "La Calle de la dirección es obligatoria."; return false; }

            if (string.IsNullOrWhiteSpace(obj.Altura) || !Regex.IsMatch(obj.Altura, @"^\d+$"))
            { mensaje = "La Altura de la dirección es obligatoria y debe contener solo números."; return false; }

            // Validación de Edad razonable
            if (obj.FechaNacimiento.HasValue)
            {
                int edad = DateTime.Now.Year - obj.FechaNacimiento.Value.Year;
                if (edad > 100 || obj.FechaNacimiento.Value.Date > DateTime.Now.Date)
                {
                    mensaje = "La fecha de nacimiento ingresada no es válida.";
                    return false;
                }
            }

            var listaExistente = _clienteDatos.Listar();
            if (listaExistente.Any(c => c.Dni == obj.Dni && c.IdCliente != obj.IdCliente))
            { mensaje = "El DNI ingresado ya pertenece a otro cliente en el sistema."; return false; }

            if (listaExistente.Any(c => c.CuilCuit == obj.CuilCuit && c.IdCliente != obj.IdCliente))
            { mensaje = "El CUIL/CUIT ingresado ya se encuentra registrado en otro cliente."; return false; }

            if (listaExistente.Any(c => c.Email.ToLower() == obj.Email.ToLower() && c.IdCliente != obj.IdCliente))
            { mensaje = "El Email ingresado ya se encuentra registrado en el sistema."; return false; }

            if (obj.IdCliente == 0)
                return _clienteDatos.Insertar(obj);
            else
                return _clienteDatos.Editar(obj);
        }

            mensaje = string.Empty;
            return _clienteDatos.Eliminar(id);

        private UsuarioDatos _usuarioDatos = new UsuarioDatos();

        public Usuario Login(string email, string password, out string mensaje)
        {
            mensaje = string.Empty;
            if (string.IsNullOrEmpty(email) || string.IsNullOrEmpty(password))
            {
                mensaje = "Debe ingresar el correo y la contraseña.";
                return null;
            }

            Usuario usuarioEncontrado = _usuarioDatos.Autenticar(email, password);
            if (usuarioEncontrado == null) mensaje = "Correo o contraseña incorrectos, o el usuario está inactivo.";
            return usuarioEncontrado;
        }

        public List<ComboUsuario> ObtenerPerfiles() => _usuarioDatos.ObtenerPerfiles();
        public List<ComboUsuario> ObtenerSucursales() => _usuarioDatos.ObtenerSucursales();
        public List<Usuario> Listar(string filtroDni = "") => _usuarioDatos.Listar(filtroDni);

        public bool Guardar(Usuario obj, out string mensaje)
        {
            mensaje = string.Empty;

            if (string.IsNullOrWhiteSpace(obj.Dni) || !Regex.IsMatch(obj.Dni, @"^\d{7,8}$"))
            { mensaje = "El DNI es obligatorio y debe contener entre 7 y 8 números."; return false; }

            if (string.IsNullOrWhiteSpace(obj.Nombre) || obj.Nombre.Trim().Length < 2)
            { mensaje = "El Nombre es obligatorio y debe tener al menos 2 letras."; return false; }

            if (string.IsNullOrWhiteSpace(obj.Apellido) || obj.Apellido.Trim().Length < 2)
            { mensaje = "El Apellido es obligatorio y debe tener al menos 2 letras."; return false; }

            if (string.IsNullOrWhiteSpace(obj.Email) || !Regex.IsMatch(obj.Email, @"^[^@\s]+@[^@\s]+\.[^@\s]+$"))
            { mensaje = "Debe ingresar un Email válido (ejemplo@dominio.com)."; return false; }

            if (string.IsNullOrWhiteSpace(obj.Telefono) || !Regex.IsMatch(obj.Telefono, @"^\d{6,15}$"))
            { mensaje = "El Teléfono es obligatorio y debe contener solo números."; return false; }

            if (string.IsNullOrWhiteSpace(obj.Calle))
            { mensaje = "La Calle de la dirección es obligatoria."; return false; }

            if (string.IsNullOrWhiteSpace(obj.Altura) || !Regex.IsMatch(obj.Altura, @"^\d+$"))
            { mensaje = "La Altura de la dirección es obligatoria y debe contener solo números."; return false; }

            if (obj.IdPerfil <= 0) { mensaje = "Debe seleccionar un Perfil válido."; return false; }
            if (obj.IdSucursal <= 0) { mensaje = "Debe seleccionar una Sucursal válida."; return false; }

            // Validación de Edad razonable
            if (obj.FechaNacimiento.HasValue)
            {
                int edad = DateTime.Now.Year - obj.FechaNacimiento.Value.Year;
                if (edad < 18 || edad > 100)
                {
                    mensaje = "El usuario debe ser mayor de edad y tener una fecha de nacimiento válida.";
                    return false;
                }
            }

            var listaExistente = _usuarioDatos.Listar();
            if (listaExistente.Any(u => u.Dni == obj.Dni && u.IdUsuario != obj.IdUsuario))
            { mensaje = "El DNI ingresado ya pertenece a otro usuario en el sistema."; return false; }

            if (listaExistente.Any(u => u.Email.ToLower() == obj.Email.ToLower() && u.IdUsuario != obj.IdUsuario))
            { mensaje = "El Email ingresado ya se encuentra registrado en el sistema."; return false; }

            if (obj.IdUsuario == 0)
            {
                if (string.IsNullOrWhiteSpace(obj.Password) || obj.Password.Length < 4)
                { mensaje = "La Contraseña es obligatoria para nuevos usuarios (mínimo 4 caracteres)."; return false; }

                // Hasheamos la contraseña antes de insertar
                obj.Password = BCrypt.Net.BCrypt.HashPassword(obj.Password);
                return _usuarioDatos.Insertar(obj);
            }
            else
            {
                if (!string.IsNullOrWhiteSpace(obj.Password))
                {
                    if (obj.Password.Length < 4)
                    {
                        mensaje = "La nueva contraseña debe tener al menos 4 caracteres.";
                        return false;
                    }
                    // Si completó el campo al editar, la hasheamos; si lo dejó vacío, Editar no lo toma.
                    obj.Password = BCrypt.Net.BCrypt.HashPassword(obj.Password);
                }
                return _usuarioDatos.Editar(obj);
            }
        }

        public bool Eliminar(int id, out string mensaje)
        {
            mensaje = string.Empty;

            // Validación: Evitar auto-eliminación
            if (id == SesionActual.IdUsuario)
            {
                mensaje = "No puede eliminar su propio usuario mientras tiene la sesión iniciada.";
                return false;
            }

            return _usuarioDatos.Eliminar(id);
        }

    public static class GeneradorFactura
    {
        public static void GenerarPDF(int idVenta, string fecha, string cliente, string cajero, decimal total, string metodoPago, List<DetalleHistorial> detalles, bool abrirAutomatico = false)
        {
            SaveFileDialog sfd = new SaveFileDialog();
            sfd.Filter = "Archivo PDF (*.pdf)|*.pdf";
            sfd.FileName = $"Factura_Aorus_Nro{idVenta}.pdf";

            if (sfd.ShowDialog() == DialogResult.OK)
            {
                try
                {
                    // Configuramos el documento tamaño A4
                    Document doc = new Document(PageSize.A4, 40, 40, 40, 40);
                    PdfWriter.GetInstance(doc, new FileStream(sfd.FileName, FileMode.Create));
                    doc.Open();

                    // Título principal
                    Font fontTitulo = FontFactory.GetFont(FontFactory.HELVETICA_BOLD, 22, BaseColor.DARK_GRAY);
                    Paragraph titulo = new Paragraph("AORUS MARKET - FACTURA", fontTitulo);
                    titulo.Alignment = Element.ALIGN_CENTER;
                    doc.Add(titulo);
                    doc.Add(new Chunk("\n\n"));

                    // Información de cabecera
                    Font fontNormal = FontFactory.GetFont(FontFactory.HELVETICA, 11);
                    Font fontBold = FontFactory.GetFont(FontFactory.HELVETICA_BOLD, 11);

                    doc.Add(new Paragraph($"Ticket N°: {idVenta}", fontBold));
                    doc.Add(new Paragraph($"Fecha: {fecha}", fontNormal));
                    doc.Add(new Paragraph($"Cliente: {cliente}", fontNormal));
                    doc.Add(new Paragraph($"Cajero: {cajero}", fontNormal));
                    doc.Add(new Paragraph($"Método de Pago: {metodoPago}", fontNormal));
                    doc.Add(new Chunk("\n\n"));

                    // Tabla de productos
                    PdfPTable tabla = new PdfPTable(4);
                    tabla.WidthPercentage = 100;
                    tabla.SetWidths(new float[] { 10f, 50f, 20f, 20f }); // Porcentajes de columnas

                    // Encabezados de tabla
                    tabla.AddCell(new PdfPCell(new Phrase("Cant.", fontBold)) { BackgroundColor = BaseColor.LIGHT_GRAY });
                    tabla.AddCell(new PdfPCell(new Phrase("Descripción", fontBold)) { BackgroundColor = BaseColor.LIGHT_GRAY });
                    tabla.AddCell(new PdfPCell(new Phrase("P. Unit", fontBold)) { BackgroundColor = BaseColor.LIGHT_GRAY });
                    tabla.AddCell(new PdfPCell(new Phrase("Subtotal", fontBold)) { BackgroundColor = BaseColor.LIGHT_GRAY });

                    // Llenar datos
                    foreach (var item in detalles)
                    {
                        tabla.AddCell(item.Cantidad.ToString());
                        tabla.AddCell(item.NombreProducto);
                        tabla.AddCell("$" + item.PrecioUnitario.ToString("0.00"));
                        tabla.AddCell("$" + item.SubTotal.ToString("0.00"));
                    }
                    doc.Add(tabla);
                    doc.Add(new Chunk("\n"));

                    // Total final
                    Font fontTotal = FontFactory.GetFont(FontFactory.HELVETICA_BOLD, 16, BaseColor.RED);
                    Paragraph pTotal = new Paragraph($"TOTAL: ${total.ToString("0.00")}", fontTotal);
                    pTotal.Alignment = Element.ALIGN_RIGHT;
                    doc.Add(pTotal);

                    doc.Close();

                    // Mostrar mensaje o abrir el PDF directamente
                    if (abrirAutomatico)
                        Process.Start(sfd.FileName);
                    else
                        MessageBox.Show("Factura PDF descargada exitosamente.", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                }
                catch (Exception ex)
                {
                    MessageBox.Show("Error al generar PDF: " + ex.Message, "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
        }
    }
