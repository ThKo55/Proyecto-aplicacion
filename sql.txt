/****** Objeto: StoredProcedure [dbo].[SP_AprobarYEjecutarBackup] ******/
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
CREATE PROCEDURE [dbo].[SP_AprobarYEjecutarBackup]
    @IdSolicitud INT,
    @AprobadorId INT,
    @RutaArchivo NVARCHAR(255)
AS
BEGIN
    SET NOCOUNT ON;
    
    DECLARE @SolicitanteId INT;
    DECLARE @EstadoActual NVARCHAR(50);
    
    SELECT @SolicitanteId = id_usuario_solicitante, @EstadoActual = estado 
    FROM historial_backups 
    WHERE id_backup = @IdSolicitud;
    
    IF @EstadoActual <> 'Pendiente'
    BEGIN
        RAISERROR('La solicitud ya fue procesada o no está pendiente.', 16, 1);
        RETURN;
    END

    IF @SolicitanteId = @AprobadorId
    BEGIN
        RAISERROR('Claves cruzadas requeridas: El aprobador debe ser un usuario distinto al solicitante.', 16, 1);
        RETURN;
    END
    
    BEGIN TRY
        DECLARE @BackupCmd NVARCHAR(MAX) = 
            'BACKUP DATABASE [SuperMercadoDB] TO DISK = ''' + @RutaArchivo + ''' WITH INIT, STATS = 10';
            
        EXEC sp_executesql @BackupCmd;
        
        UPDATE historial_backups
        SET id_usuario_aprobador = @AprobadorId,
            estado = 'Completado',
            fecha_ejecucion = GETDATE(),
            ruta_archivo = @RutaArchivo
        WHERE id_backup = @IdSolicitud;
        
    END TRY
    BEGIN CATCH
        UPDATE historial_backups
        SET estado = 'Fallido',
            mensaje_error = ERROR_MESSAGE()
        WHERE id_backup = @IdSolicitud;
        
        DECLARE @ErrorMsg NVARCHAR(MAX) = ERROR_MESSAGE();
        RAISERROR(@ErrorMsg, 16, 1);
    END CATCH
END;
GO

/****** Objeto: StoredProcedure [dbo].[SP_HistorialBackupsRealizados] ******/
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
CREATE PROCEDURE [dbo].[SP_HistorialBackupsRealizados]
AS
BEGIN
    SET NOCOUNT ON;
    
    SELECT 
        h.id_backup AS Id,
        us.nombre + ' ' + us.apellido AS Solicitante,
        ua.nombre + ' ' + ua.apellido AS Aprobador,
        h.fecha_ejecucion AS FechaEjecucion,
        h.ruta_archivo AS RutaArchivo,
        h.estado AS Estado,
        h.mensaje_error AS MensajeError
    FROM historial_backups h
    INNER JOIN usuario us ON h.id_usuario_solicitante = us.id_usuario
    LEFT JOIN usuario ua ON h.id_usuario_aprobador = ua.id_usuario
    WHERE h.estado IN ('Completado', 'Fallido')
    ORDER BY h.fecha_ejecucion DESC;
END;
GO

/****** Objeto: StoredProcedure [dbo].[SP_ListarBackupsPendientes] ******/
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
CREATE PROCEDURE [dbo].[SP_ListarBackupsPendientes]
AS
BEGIN
    SET NOCOUNT ON;
    
    SELECT 
        h.id_backup AS Id,
        u.nombre + ' ' + u.apellido AS Solicitante,
        h.fecha_solicitud AS FechaSolicitud,
        h.fecha_programada AS FechaProgramada,
        h.estado AS Estado
    FROM historial_backups h
    INNER JOIN usuario u ON h.id_usuario_solicitante = u.id_usuario
    WHERE h.estado = 'Pendiente'
    ORDER BY h.fecha_programada ASC;
END;
GO

/****** Objeto: StoredProcedure [dbo].[SP_RegistrarSolicitudBackup] ******/
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
CREATE PROCEDURE [dbo].[SP_RegistrarSolicitudBackup]
    @SolicitanteId INT,
    @FechaProgramada DATETIME
AS
BEGIN
    SET NOCOUNT ON;
    
    INSERT INTO historial_backups (id_usuario_solicitante, estado, fecha_solicitud, fecha_programada)
    VALUES (@SolicitanteId, 'Pendiente', GETDATE(), @FechaProgramada);
    
    SELECT SCOPE_IDENTITY() AS NuevoId;
END;
GO
